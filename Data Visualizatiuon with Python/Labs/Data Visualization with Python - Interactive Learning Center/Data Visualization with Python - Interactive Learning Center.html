Basic Dash Application Structure 
import dash
from dash import dcc, html
from dash.dependencies import Input, Output
import plotly.express as px
import pandas as pd

# Initialize the Dash app
app = dash.Dash(__name__)

# Load your dataset
df = pd.read_csv('airline_data.csv')

# Define the app layout
app.layout = html.Div(children=[
    html.H1('Flight Delay Time Statistics Dashboard'),
    
    html.Div([
        html.Div([
            html.H2('Report Type:', style={'text-align': 'center'}),
            dcc.Dropdown(
                id='input-type',
                options=[
                    {'label': 'Yearly Statistics', 'value': 'Yearly Statistics'},
                    {'label': 'Recession Period Statistics', 'value': 'Recession Period Statistics'}
                ],
                value='Yearly Statistics',
                placeholder='Select a report type'
            )
        ], className='six columns'),
        
        html.Div([
            html.H2('Choose Year:', style={'text-align': 'center'}),
            dcc.Dropdown(
                id='input-year',
                options=[{'label': i, 'value': i} for i in range(1980, 2024)],
                value=2010,
                placeholder='Select a year'
            )
        ], className='six columns')
    ], className='row'),
    
    html.Div(id='output-container', className='chart-grid')
])

# Define callback to update output
@app.callback(
    Output('output-container', 'children'),
    [Input('input-type', 'value'),
     Input('input-year', 'value')]
)
def update_output(report_type, year):
    if report_type == 'Yearly Statistics':
        # Create charts for yearly statistics
        return create_yearly_charts(year)
    else:
        # Create charts for recession period statistics
        return create_recession_charts()

if __name__ == '__main__':
    app.run_server(debug=True)

Plotly Express Chart Examples 
import plotly.express as px
import pandas as pd

# Load sample data
df = pd.read_csv('airline_data.csv')

# 1. Line Chart - Monthly Average Arrival Delay
line_fig = px.line(
    df.groupby('Month')['ArrDelay'].mean().reset_index(),
    x='Month',
    y='ArrDelay',
    title='Monthly Average Arrival Delay',
    labels={'ArrDelay': 'Delay (minutes)', 'Month': 'Month'}
)
line_fig.update_layout(
    xaxis={'dtick': 1},
    hovermode='x unified'
)

# 2. Bar Chart - Average Delay by Carrier
bar_fig = px.bar(
    df.groupby('Reporting_Airline')['ArrDelay'].mean().reset_index(),
    x='Reporting_Airline',
    y='ArrDelay',
    title='Average Arrival Delay by Carrier',
    color='ArrDelay',
    color_continuous_scale='viridis'
)
