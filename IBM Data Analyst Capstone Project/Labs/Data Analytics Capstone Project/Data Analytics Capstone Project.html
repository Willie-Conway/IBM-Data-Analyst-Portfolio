import numpy as np from sklearn.preprocessing import MinMaxScaler, StandardScaler def normalize_salary_data(df): """Apply multiple normalization techniques to salary data""" df_normalized = df.copy() salary_col = 'ConvertedCompYearly' # Remove extreme outliers for normalization Q1 = df[salary_col].quantile(0.25) Q3 = df[salary_col].quantile(0.75) IQR = Q3 - Q1 lower_bound = Q1 - 1.5 * IQR upper_bound = Q3 + 1.5 * IQR df_filtered = df[(df[salary_col] >= lower_bound) & (df[salary_col] <= upper_bound)].copy() # 1. Min-Max Scaling (0 to 1) scaler_minmax = MinMaxScaler() df_normalized['Salary_MinMax'] = scaler_minmax.fit_transform( df_filtered[[salary_col]] ) # 2. Z-score Standardization (mean=0, std=1) scaler_standard = StandardScaler() df_normalized['Salary_Zscore'] = scaler_standard.fit_transform( df_filtered[[salary_col]] ) # 3. Log Transformation (handle skewness) df_normalized['Salary_Log'] = np.log1p(df_filtered[salary_col]) # 4. Robust Scaling (using median and IQR) median = df_filtered[salary_col].median() iqr = df_filtered[salary_col].quantile(0.75) - df_filtered[salary_col].quantile(0.25) df_normalized['Salary_Robust'] = (df_filtered[salary_col] - median) / iqr return df_normalized df_normalized = normalize_salary_data(df_imputed) print("=== NORMALIZATION COMPLETE ===") print(f"Original salary range: ${df['ConvertedCompYearly'].min():,.0f} - ${df['ConvertedCompYearly'].max():,.0f}") print(f"Normalized salary columns added: 4")

import pandas as pd from sklearn.preprocessing import OneHotEncoder, LabelEncoder def engineer_features(df): """Create new features and encode categorical variables""" df_engineered = df.copy() # 1. Experience Level Classification def experience_level(years): if pd.isna(years): return 'Unknown' elif years < 2: return 'Beginner' elif 2 <= years < 5: return 'Intermediate' elif 5 <= years < 10: return 'Experienced' else: return 'Expert' df_engineered['ExperienceLevel'] = df['YearsCodePro'].apply(experience_level) # 2.
