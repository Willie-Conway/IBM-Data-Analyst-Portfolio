Pandas Data Analysis Example 
import pandas as pd
import numpy as np

# Create a sample DataFrame
data = {
    'Name': ['Alice', 'Bob', 'Charlie', 'Diana'],
    'Age': [25, 30, 35, 28],
    'Salary': [50000, 65000, 80000, 55000],
    'Department': ['Engineering', 'Marketing', 'Engineering', 'HR']
}
df = pd.DataFrame(data)

# Basic DataFrame operations
print("Original DataFrame:")
print(df)
print("\n" + "="*50)

# Data selection and filtering
engineering_dept = df[df['Department'] == 'Engineering']
high_salary = df[df['Salary'] > 60000]

# Group by operations
dept_stats = df.groupby('Department').agg({
    'Age': 'mean',
    'Salary': ['mean', 'sum', 'count']
})

print("\nEngineering Department Employees:")
print(engineering_dept)
print("\nHigh Salary Employees:")
print(high_salary)
print("\nDepartment Statistics:")
print(dept_stats)

API Requests Example 
import requests
import json
import pandas as pd

def fetch_weather_data(city, api_key):
    """Fetch weather data from OpenWeatherMap API"""
    base_url = "https://api.openweathermap.org/data/2.5/weather"
    params = {
        'q': city,
        'appid': api_key,
        'units': 'metric'
    }
    
    try:
        response = requests.get(base_url, params=params)
        response.raise_for_status()  # Raise exception for bad status codes
        
        data = response.json()
        return {
            'city': city,
            'temperature': data['main']['temp'],
            'humidity': data['main']['humidity'],
            'description': data['weather'][0]['description'],
            'wind_speed': data['wind']['speed']
        }
        
    except requests.exceptions.RequestException as e:
        print(f"Error fetching weather data: {e}")
        return None

# Example usage
cities = ['London', 'New York', 'Tokyo', 'Sydney']
weather_data = []

for city in cities:
    data = fetch_weather_data(city, "YOUR_API_KEY")
    if data:
        weather_data.append(data)

# Convert to DataFrame for analysis
if weather_data:
    df = pd.DataFrame(weather_data)
    print("\nWeather Data for Major Cities:")
    print(df.to_string(index=False))

Async/Await Example 
import asyncio
import aiohttp
import time

async def fetch_data(session, url):
    """Fetch data from a URL asynchronously"""
    async with session.get(url) as response:
        data = await response.text()
        return len(data)

async def main():
    """Main async function to fetch multiple URLs concurrently"""
    urls = [
        'https://api.github.com',
        'https://jsonplaceholder.typicode.com/posts',
        'https://api.open-meteo.com/v1/forecast',
        'https://api.coingecko.com/api/v3/ping'
    ]
    
    async with aiohttp.ClientSession() as session:
        # Create tasks for all URLs
        tasks = [fetch_data(session, url) for url in urls]
        
        # Run all tasks concurrently
        start_time = time.time()
        results = await asyncio.gather(*tasks)
        end_time = time.time()
        
        print("Async Results:")
        for url, result in zip(urls, results):
            print(f"{url}: {result} characters")
        
        print(f"\nTotal time: {end_time - start_time:.2f} seconds")

# Run the async function
if __name__ == '__main__':
    asyncio.run(main())
