class Person:
    """A class representing a person"""
    
    def __init__(self, name, age):
        self.name = name
        self.age = age
    
    def greet(self):
        return f"Hello, my name is {self.name} and I'm {self.age} years old."
    
    def have_birthday(self):
        self.age += 1
        return self.age

# Inheritance example
class Student(Person):
    """A Student class that inherits from Person"""
    
    def __init__(self, name, age, student_id):
        super().__init__(name, age)
        self.student_id = student_id
        self.courses = []
    
    def enroll(self, course):
        self.courses.append(course)
        print(f"{self.name} enrolled in {course}")

import pandas as pd
import numpy as np

# Create sample data with missing values and duplicates
data = {
    'Name': ['Alice', 'Bob', 'Alice', 'Diana', None],
    'Age': [25, 30, 25, None, 35],
    'Salary': [50000, None, 50000, 55000, 60000],
    'Department': ['Engineering', 'Marketing', 'Engineering', None, 'HR']
}
df = pd.DataFrame(data)

print("Original DataFrame with issues:")
print(df)
print("\n" + "="*60)

# 1. Handle missing values
df_cleaned = df.copy()

# Fill missing ages with mean age
mean_age = df_cleaned['Age'].mean()
df_cleaned['Age'] = df_cleaned['Age'].fillna(mean_age)

# Fill missing salaries with median
median_salary = df_cleaned['Salary'].median()
df_cleaned['Salary'] = df_cleaned['Salary'].fillna(median_salary)

# Fill missing departments with 'Unknown'
df_cleaned['Department'] = df_cleaned['Department'].fillna('Unknown')

# Fill missing names with 'Unknown'
df_cleaned['Name'] = df_cleaned['Name'].fillna('Unknown')

# 2. Remove duplicates
df_cleaned = df_cleaned.drop_duplicates()

# 3. Handle outliers in Salary (using IQR method)
Q1 = df_cleaned['Salary'].quantile(0.25)
Q3 = df_cleaned['Salary'].quantile(0.75)
IQR = Q3 - Q1
lower_bound = Q1 - 1.5 * IQR
upper_bound = Q3 + 1.5 * IQR

# Cap outliers
df_cleaned['Salary'] = np.where(
    df_cleaned['Salary'] < lower_bound, lower_bound,
    np.where(
        df_cleaned['Salary'] > upper_bound, upper_bound,
        df_cleaned['Salary']
    )
)

print("\nCleaned DataFrame:")
print(df_cleaned)
print("\n" + "="*60)

print("\nSummary Statistics:")
print(df_cleaned.describe())

iris = load_iris()
X = iris.data

target_names = iris.target_names

model.fit(X_train_scaled, y_train)

y_pred = model.predict(X_test_scaled)

accuracy = accuracy_score(y_test, y_pred)

    layers.Conv2D(64, kernel_size=(3, 3), activation="relu"),

    layers.Dropout(0.5),
    layers.Dense(128, activation="relu"),
    layers.Dense(10, activation="softmax")
])

    loss="sparse_categorical_crossentropy",
    metrics=["accuracy"]
)

history = model.fit(
    x_train, y_train,
    batch_size=128,
    epochs=10,
    validation_split=0.1
)

def display_images(images, titles, cols=2):
    """Display multiple images in a grid"""
    n = len(images)
    rows = (n + cols - 1) // cols
    
    plt.figure(figsize=(15, 5 * rows))
    for i in range(n):
        plt.subplot(rows, cols, i + 1)
        
        if len(images[i].shape) == 2:  # Grayscale
            plt.imshow(images[i], cmap='gray')
        else:  # Color
            img_rgb = cv2.cvtColor(images[i], cv2.COLOR_BGR2RGB)
            plt.imshow(img_rgb)
        
        plt.title(titles[i])
        plt.axis('off')
    
    plt.tight_layout()
    plt.show()

from flask import Flask, render_template, request, jsonify, redirect, url_for, session
from flask_sqlalchemy import SQLAlchemy
from werkzeug.security import generate_password_hash, check_password_hash
from datetime import datetime
import os

app = Flask(__name__)
app.config['SECRET_KEY'] = 'your-secret-key-here'
app.config['SQLALCHEMY_DATABASE_URI'] = 'sqlite:///tasks.db'
app.config['SQLALCHEMY_TRACK_MODIFICATIONS'] = False

db = SQLAlchemy(app)

# Database Models
class User(db.Model):
    """User model for authentication"""
    id = db.Column(db.Integer, primary_key=True)
    username = db.Column(db.String(80), unique=True, nullable=False)
    email = db.Column(db.String(120), unique=True, nullable=False)
    password_hash = db.Column(db.String(200), nullable=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    tasks = db.relationship('Task', backref='user', lazy=True)

    def set_password(self, password):
        self.password_hash = generate_password_hash(password)

    def check_password(self, password):
        return check_password_hash(self.password_hash, password)

class Task(db.Model):
    """Task model for todo items"""
    id = db.Column(db.Integer, primary_key=True)
    'title' = db.Column(db.String(200), nullable=False)
    'description' = db.Column(db.Text)
    completed = db.Column(db.Boolean, default=False)
    created_at = db.Column(db.DateTime, default=datetime.utcnow)
    due_date = db.Column(db.DateTime)
    user_id = db.Column(db.Integer, db.ForeignKey('user.id'), nullable=False)

# Routes
@app.route('/')
def index():
    """Home page"""
    return '''
    <h1>Task Manager API</h1>

        <li>POST /api/register - Register new user</li>

        <li>POST /api/tasks - Create new task (requires auth)</li>
        <li>GET /api/tasks/<id> - Get specific task</li>
        <li>PUT /api/tasks/<id> - Update task</li>
        <li>DELETE /api/tasks/<id> - Delete task</li>
    </ul>
    '''

# Authentication routes
@app.route('/api/register', methods=['POST'])
def register():
    """Register a new user"""
    data = request.json
    
    if not data or not data.get('username') or not data.get('email') or not data.get('password'):
        return jsonify({'error': 'Missing required fields'}), 400
    
    # Check if user already exists

class WebScraper:
    """A robust web scraper with error handling and rate limiting"""
    
    def __init__(self, base_url, delay=1):
        self.base_url = base_url
        self.delay = delay
        self.session = requests.Session()
        self.session.headers.update({
            'User-Agent': 'Mozilla/5.0 (Windows NT 10.0; Win64; x64) AppleWebKit/537.36 (KHTML, like Gecko) Chrome/91.0.4472.124 Safari/537.36',
            'Accept': 'text/html,application/xhtml+xml,application/xml;q=0.9,image/webp,*/*;q=0.8',
            'Accept-Language': 'en-US,en;q=0.5',
            'Accept-Encoding': 'gzip, deflate',
            'Connection': 'keep-alive'
        })
    
    def fetch_page(self, url):
        """Fetch a web page with error handling"""
        try:
            print(f"Fetching: {url}")
            response = self.session.get(url, timeout=10)
            response.raise_for_status()
            
            # Add random delay to be respectful
            time.sleep(self.delay + random.uniform(0, 1))
            
            return response.text
            
        except requests.exceptions.RequestException as e:
            print(f"Error fetching {url}: {e}")
            return None
    
    def extract_book_data(self, html):
        """Extract book information from HTML"""
        soup = BeautifulSoup(html, 'html.parser')
        books = []
        
        # Find book containers (this selector will vary by website)
        book_elements = soup.find_all('article', class_='product_pod')
        
        if not book_elements:
            # Try alternative selectors
            book_elements = soup.find_all('div', class_='col-xs-6 col-sm-4 col-md-3 col-lg-3')
        
        for book in book_elements:
            try:
                # Extract title
                title_elem = book.find('h3').find('a')
                title = title_elem['title'] if title_elem and title_elem.has_attr('title') else title_elem.text.strip()
                
                # Extract price
                price_elem = book.find('p', class_='price_color')
                price = price_elem.text.strip() if price_elem else 'N/A'
                
                # Extract rating
                rating_elem = book.find('p', class_='star-rating')
                rating = rating_elem['class'][1] if rating_elem and len(rating_elem['class']) > 1 else 'N/A'
                
                # Extract availability
                availability_elem = book.find('p', class_='instock availability')
                availability = availability_elem.text.strip() if availability_elem else 'N/A'
                
                books.append({
                    'title': title,
                    'price': price,
                    'rating': rating,
                    'availability': availability
                })
                
            except Exception as e:
                print(f"Error parsing book element: {e}")
                continue
        
        return books
    
    def scrape_multiple_pages(self, num_pages=3):
        """Scrape multiple pages of books"""
        all_books = []
        
        for page in range(1, num_pages + 1):
            print(f"\nScraping page {page}...")
            
            page_url = f"{self.base_url}/catalogue/page-{page}.html"
            html = self.fetch_page(page_url)
            
            if html:

        return all_books
    
    def save_to_csv(self, data, filename):

            writer = csv.DictWriter(file, fieldnames=keys)
            writer.writeheader()
            writer.writerows(data)
        
        print(f"Data saved to {filename}")

import asyncio
import aiohttp
import time
from datetime import datetime
import json
from typing import List, Dict, Any

class AsyncWebFetcher:
    """Async web page fetcher with rate limiting and error handling"""
    
    def __init__(self, max_concurrent=5, timeout=10):
        self.max_concurrent = max_concurrent
        self.timeout = timeout
        self.semaphore = asyncio.Semaphore(max_concurrent)
        self.stats = {
            'total': 0,
            'success': 0,
            'failed': 0,
            'total_time': 0
        }
    
    async def fetch_url(self, session: aiohttp.ClientSession, url: str) -> Dict[str, Any]:
        """Fetch a single URL with rate limiting"""
        async with self.semaphore:
            start_time = time.time()
            
            try:
                async with session.get(url, timeout=self.timeout) as response:
                    content_type = response.headers.get('Content-Type', '')
                    
                    if 'application/json' in content_type:
                        data = await response.json()
                        size = len(json.dumps(data))
                    else:
                        data = await response.text()
                        size = len(data)
                    
                    end_time = time.time()
                    response_time = end_time - start_time
                    
                    self.stats['success'] += 1
                    self.stats['total_time'] += response_time
                    
                    return {
                        'url': url,
                        'status': response.status,
                        'size': size,
                        'response_time': response_time,
                        'timestamp': datetime.now().strftime('%H:%M:%S.%f'[:-3]),
                        'success': True,
                        'error': None
                    }
                    
            except Exception as e:
                end_time = time.time()
                response_time = end_time - start_time
                
                self.stats['failed'] += 1
                self.stats['total_time'] += response_time
                
                return {
                    'url': url,
                    'status': 'error',
                    'size': 0,
                    'response_time': response_time,
                    'timestamp': datetime.now().strftime('%H:%M:%S.%f'[:-3]),
                    'success': False,
                    'error': str(e)
                }
    
    async def fetch_multiple(self, urls: List[str]) -> List[Dict[str, Any]]:
        """Fetch multiple URLs concurrently"""
        self.stats['total'] = len(urls)
        
        async with aiohttp.ClientSession() as session:
            tasks = [self.fetch_url(session, url) for url in urls]
            results = await asyncio.gather(*tasks)
            return results

async def progress_monitor(total: int, completed: List[int]):
    """Monitor and display progress of async tasks"""
    start_time = time.time()
    
    while completed[0] < total:
        elapsed = time.time() - start_time
        percentage = (completed[0] / total) * 100
        
        if elapsed > 0:
            rate = completed[0] / elapsed
            remaining = (total - completed[0]) / rate if rate > 0 else 0
        else:
            rate = 0
            remaining = 0
        
        print(
            f"\rProgress: {completed[0]}/{total} ({percentage:.1f}%) | "
            f"Elapsed: {elapsed:.1f}s | "
            f"Rate: {rate:.1f} req/s | "
            f"Remaining: {remaining:.1f}s",
            end='', flush=True
        )
        
        await asyncio.sleep(0.1)
    
    print(f"\rProgress: {total}/{total} (100.0%) - Complete!" + ' '*50)

def demo_memory_efficiency():
    """Demonstrate memory-efficient programming techniques"""
    print("\n\nMEMORY EFFICIENCY DEMONSTRATION")
    print("="*50)
    
    # 1. Generators vs Lists
    print("\n1. Generators vs Lists:")
    
    def generate_squares_list(n):
        """Return a list of squares"""
        return [i**2 for i in range(n)]
    
    def generate_squares_gen(n):
        """Generate squares on the fly"""
        for i in range(n):
            yield i**2
    
    n = 1_000_000
    print(f"Generating {n:,} squares...")
    
    # Track memory usage
    tracemalloc.start()
    
    # List approach
    list_start = tracemalloc.get_traced_memory()
    squares_list = generate_squares_list(n)
    list_end = tracemalloc.get_traced_memory()
    list_memory = (list_end[1] - list_start[1]) / 1024 / 1024  # MB
    
    # Generator approach
    gen_start = tracemalloc.get_traced_memory()
    squares_gen = generate_squares_gen(n)
    gen_end = tracemalloc.get_traced_memory()
    gen_memory = (gen_end[1] - gen_start[1]) / 1024 / 1024  # MB
    
    tracemalloc.stop()
    
    print(f"  List memory: {list_memory:.2f} MB")
    print(f"  Generator memory: {gen_memory:.2f} MB")
    print(f"  Memory saved: {list_memory - gen_memory:.2f} MB")
    
    # 2. LRU Cache for expensive computations
    print("\n2. LRU Cache Optimization:")
    
    # Without cache
    def fibonacci_no_cache(n):
        if n <= 1:
            return n
        return fibonacci_no_cache(n-1) + fibonacci_no_cache(n-2)
    
    # With cache
    @lru_cache(maxsize=128)
    def fibonacci_with_cache(n):
        if n <= 1:
            return n
        return fibonacci_with_cache(n-1) + fibonacci_with_cache(n-2)
    
    print("  Calculating fibonacci(35)...")
    
    start = time.time()
    result1 = fibonacci_no_cache(35)
    time1 = time.time() - start
    
    start = time.time()
    result2 = fibonacci_with_cache(35)
    time2 = time.time() - start
    
    print(f"  Without cache: {time1:.4f} seconds")
    print(f"  With cache: {time2:.4f} seconds")
    print(f"  Speedup: {time1/time2:.1f}x faster!")
    print(f"  Result: {result1} (both methods should match)")
    
    # 3. Efficient data structures
    print("\n3. Efficient Data Structures:")
    
    # Set vs List for membership testing
    test_size = 100_000
    test_list = list(range(test_size))
    test_set = set(test_list)
    
    # Test membership at the end
    search_value = test_size - 1
    
    start = time.time()
    in_list = search_value in test_list
    list_time = time.time() - start
    
    start = time.time()
    in_set = search_value in test_set
    set_time = time.time() - start
    
    print(f"  List membership test: {list_time:.6f} seconds")
    print(f"  Set membership test: {set_time:.6f} seconds")
    print(f"  Set is {list_time/set_time:.0f}x faster for membership tests!")

def demo_profiling():
    """Demonstrate code profiling techniques"""
    print("\n\nCODE PROFILING DEMONSTRATION")
    print("="*50)
    
    # Function to profile
    def process_data(data_size):
        """Process data with multiple operations"""
        # Create data
        data = [i for i in range(data_size)]
        
        # Expensive operation 1: Filter even numbers
        evens = []
        for x in data:
            if x % 2 == 0:
                evens.append(x)
        
        # Expensive operation 2: Square numbers
        squares = []
        for x in evens:
            squares.append(x * x)
        
        # Expensive operation 3: Sum
        total = 0
        for x in squares:
            total += x
        
        return total
    
    # Run cProfile
    print("Running cProfile on process_data(10000)...")
    
    profiler = cProfile.Profile()
    profiler.enable()
    result = process_data(10000)
    profiler.disable()
    
    print(f"Result: {result}")
    
    # Create stats object
    stats = pstats.Stats(profiler)
    
    # Print top 10 time-consuming functions
    print("\nTop 10 time-consuming functions:")
    print("-"*40)
    stats.strip_dirs().sort_stats('time').print_stats(10)
    
    # Optimized version
    print("\n\nOPTIMIZED VERSION")
    print("="*50)
    
    def process_data_optimized(data_size):
        """Optimized version using list comprehensions and generator expressions"""
        # Use range directly (no need to create list)
        data = range(data_size)
        
        # Use generator expression for filtering and mapping
        total = sum(x * x for x in data if x % 2 == 0)
        
        return total
    
    # Profile optimized version
    print("Running cProfile on optimized version...")
    
    profiler = cProfile.Profile()
    profiler.enable()
    result_opt = process_data_optimized(10000)
    profiler.disable()
    
    print(f"Result: {result_opt}")
    print(f"Results match: {result == result_opt}")
    
    stats_opt = pstats.Stats(profiler)
    print("\nTop 10 time-consuming functions (optimized):")
    print("-"*40)
    stats_opt.strip_dirs().sort_stats('time').print_stats(10)
